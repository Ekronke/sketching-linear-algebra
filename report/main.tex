\section{Subspace Embeddings}\label{sec:embeddings}
\begin{definition}[Oblivious subspace embedding]
A distribution over $S \in \R^{m \times n}$ is an $(\varepsilon,\delta,d)$-OSE if for every
$A \in \R^{n \times d}$, with probability at least $1-\delta$,
\[
(1-\varepsilon)\|Ax\|_2 \le \|SAx\|_2 \le (1+\varepsilon)\|Ax\|_2 \quad \text{for all } x \in \R^d .
\]
\end{definition}
